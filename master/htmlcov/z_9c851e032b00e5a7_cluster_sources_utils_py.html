11def parse_cluster_sources_attribute(cube) -> dict: 

12 """Parse the cluster_sources dictionary from a cube attribute. 

13 

14 The cluster_sources attribute is typically set by RealizationClusterAndMatch 

15 and consumed by RealizationSelection, StochasticNoise, and temporal 

16 interpolation plugins to determine which forecast source (model) was used 

17 for each cluster at each forecast period. 

18 

19 Args: 

20 cube: 

21 A cube containing the cluster_sources attribute as a JSON string. 

22 

23 Returns: 

24 Dictionary mapping cluster indices (as strings) to model mappings. 

25 Format: {cluster_idx: {model_name: [forecast_periods_in_seconds]}}. 

26 If the cluster_sources attribute is not present, returns an empty dictionary. 

27 

28 Raises: 

29 ValueError: If the cluster_sources attribute is malformed JSON. 

30 ValueError: If the attribute is not a dictionary or JSON string. 

31 ValueError: If the parsed JSON is not a dictionary. 

32 """ 

33 cluster_sources_attr = cube.attributes.get("cluster_sources") 

34 if cluster_sources_attr is None: 

35 return {} 

36 

37 # Parse JSON string if needed 

38 if isinstance(cluster_sources_attr, str): 

39 try: 

40 parsed = json.loads(cluster_sources_attr) 

41 except json.JSONDecodeError as err: 

42 raise ValueError(f"Failed to parse cluster sources JSON: {err}") 

43 elif isinstance(cluster_sources_attr, dict): 

44 parsed = cluster_sources_attr 

45 else: 

46 raise ValueError( 

47 "Cluster sources attribute must be a dictionary or JSON string, " 

48 f"got {type(cluster_sources_attr)}" 

49 ) 

50 

51 if not isinstance(parsed, dict): 

52 raise ValueError("Cluster sources attribute must be a dictionary") 

53 

54 return parsed

57def find_nearest_forecast_period_gte( 

58 forecast_periods: Optional[set[int]], target_fp: int 

59) -> tuple[int, bool]: 

60 """Find the nearest forecast period >= target_fp from a set of forecast periods. 

61 

62 This is used for transition handling when querying secondary mappings: if an 

63 exact forecast period is not available, we find the next earliest one. 

64 

65 Args: 

66 forecast_periods: 

67 Set of available forecast periods (in seconds). 

68 If None or empty, returns (target_fp, False). 

69 target_fp: 

70 The target forecast period (in seconds) to find. 

71 

72 Returns: 

73 Tuple of (nearest_fp, use_secondary): 

74 - nearest_fp: The smallest forecast period from forecast_periods that is 

75 >= target_fp, or target_fp if none exist or forecast_periods is empty. 

76 - use_secondary: Boolean indicating whether the nearest_fp is from 

77 forecast_periods (True) or equals target_fp (False). 

78 """ 

79 if forecast_periods: 

80 valid_fps = [fp for fp in forecast_periods if fp >= target_fp] 

81 if valid_fps: 

82 return min(valid_fps), True 

83 return target_fp, False 

84 return target_fp, False

87def get_source_for_forecast_period( 

88 cluster_sources: dict, realization_idx: int, fp_seconds: int 

89) -> Optional[str]: 

90 """Determine which forecast source (model) is active for a given realization and 

91 forecast period. 

92 

93 Uses transition handling: if the exact forecast period is not found in 

94 cluster_sources for this realization, we look for the largest period 

95 that is still <= the target period (conservative: stay with last-known source). 

96 

97 Args: 

98 cluster_sources: 

99 Dictionary mapping cluster/realization indices (as strings or ints) 

100 to model sources. Format: {realization_idx: {model_name: [fp1, fp2, ...]}} 

101 realization_idx: 

102 The realization/cluster index to query. 

103 fp_seconds: 

104 The forecast period in seconds. 

105 

106 Returns: 

107 The name of the forecast source (model) active at this forecast period, 

108 or None if the realization is not found or no source covers this period. 

109 

110 Example: 

111 >>> sources = { 

112 ... "0": { 

113 ... "uk_ens": [3600, 21600, 43200, 86400, 129600, 172800], 

114 ... "gl_ens": [432000, 475200, 518400], 

115 ... } 

116 ... } 

117 >>> get_source_for_forecast_period(sources, "0", 90000) 

118 'uk_ens' 

119 >>> get_source_for_forecast_period(sources, "0", 432000) 

120 'gl_ens' 

121 >>> get_source_for_forecast_period( 

122 ... sources, "0", 400000 

123 ... ) # Between uk_ens end & gl_ens start 

124 'uk_ens' # Returns last-known source 

125 """ 

126 # Convert to string key in case int was provided 

127 real_key = str(realization_idx) 

128 

129 if real_key not in cluster_sources: 

130 return None 

131 

132 model_dict = cluster_sources[real_key] 

133 

134 # Exact match: check if fp_seconds is in any model's period list 

135 for model_name, periods in model_dict.items(): 

136 if fp_seconds in periods: 

137 return model_name 

138 

139 # Transition handling: find largest period <= fp_seconds (last-known source) 

140 # This ensures conservative behavior at model boundaries 

141 best_model = None 

142 best_period = -1 

143 

144 for model_name, periods in model_dict.items(): 

145 for period in periods: 

146 if period <= fp_seconds and period > best_period: 

147 best_period = period 

148 best_model = model_name 

149 

150 return best_model
